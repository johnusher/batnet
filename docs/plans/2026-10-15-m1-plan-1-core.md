## Global Constraints

- Module path `github.com/johnusher/batnet`; `go 1.22` in `go.mod`.
- `CGO_ENABLED=0` must build for `linux/arm64` and `linux/arm GOARM=6`.
- Entry key `(origin node, seq)`, seq starts at 1; ordered by HLC; signed by the origin node (ed25519); never changed.
- Inter-node traffic sealed with XChaCha20-Poly1305, key derived from the village key with HKDF-SHA256.
- Heartbeat every 5 s on a Pi (configurable; tests use 20 ms).
- Priority is a level 0–9 stored in each entry.
- Public repo: no keys, real names, places, node IDs or MACs in code, tests or fixtures (placeholders like `Anna · Miller House`).
- The spec's `internal/log/` is named `internal/ledger/` (Go's `log` package would be shadowed); `internal/sync/` is `internal/meshsync/` for the same reason.

## Review Focus

1. **A forged or foreign packet arrives** (bad signature, sealed with another village's key, truncated garbage): it is dropped silently, never stored, and nothing crashes. → Task 10, `TestRejectsForgedAndForeign`, `TestGarbageIgnored`.
2. **The same entry arrives many times** (rebroadcasts, repair overlapping live delivery): stored once, `OnEntry` called once. → Task 10, `TestOnEntryOncePerEntry`.
3. **Entries arrive out of order** (seq 5 before 4): 5 is kept, the vector stays at the contiguous point, repair fills 4. → Task 4 store suite `gap`, Task 10 lossy test.
4. **A Pi's clock is badly wrong** (1970 after a power cut, or far in the future): local events still order after everything it has seen. → Task 1, `TestWrongClockStillOrders`.
5. **batnetd restarts on an existing database:** the author continues its sequence and never reuses a seq. → Task 6, `TestAuthorResumes`.

```go
// Package hlc is a hybrid logical clock: wall-clock milliseconds plus a
// counter, so events order correctly even when a Pi's clock is wrong.
package hlc

import "sync"

// Timestamp orders events across nodes.
type Timestamp struct {
	Wall    int64  // milliseconds since the Unix epoch, as best known
	Logical uint32 // breaks ties within one millisecond
}

// Less reports whether t is before o.
func (t Timestamp) Less(o Timestamp) bool {
	if t.Wall != o.Wall {
		return t.Wall < o.Wall
	}
	return t.Logical < o.Logical
}

// Clock hands out timestamps later than every timestamp it has produced or seen.
type Clock struct {
	mu   sync.Mutex
	phys func() int64
	last Timestamp
}

// New returns a clock reading physical time from phys (milliseconds).
func New(phys func() int64) *Clock { return &Clock{phys: phys} }

// Now returns a timestamp for a local event.
func (c *Clock) Now() Timestamp {
	c.mu.Lock()
	defer c.mu.Unlock()
	if p := c.phys(); p > c.last.Wall {
		c.last = Timestamp{Wall: p}
	} else {
		c.last.Logical++
	}
	return c.last
}

// Observe merges a timestamp received from another node.
func (c *Clock) Observe(r Timestamp) {
	c.mu.Lock()
	defer c.mu.Unlock()
	if c.last.Less(r) {
		c.last = r
	}
}
```

```go
// Package ledger is batnet's replicated log: signed, append-only entries keyed
// by (origin node, seq), ordered by hybrid logical clock.
package ledger

import (
	"crypto/ed25519"
	"encoding/binary"
	"encoding/hex"
	"errors"
	"fmt"

	"github.com/johnusher/batnet/internal/hlc"
)

// NodeID is a node's ed25519 public key.
type NodeID [ed25519.PublicKeySize]byte

// NodeIDFromPublic turns a public key into a NodeID.
func NodeIDFromPublic(pub ed25519.PublicKey) NodeID {
	var id NodeID
	copy(id[:], pub)
	return id
}

// String is a short form for logs: the first 4 bytes in hex.
func (n NodeID) String() string { return hex.EncodeToString(n[:4]) }

// Key identifies an entry: who wrote it and its place in their sequence (from 1).
type Key struct {
	Origin NodeID
	Seq    uint64
}

// Type says what an entry means; its Body is JSON for that type.
type Type string

const (
	TypeNodeAnnounce   Type = "node.announce"
	TypeHouseJoin      Type = "house.join"
	TypeHouseApprove   Type = "house.approve"
	TypeAccountRequest Type = "account.request"
	TypeAccountApprove Type = "account.approve"
	TypeAccountDeny    Type = "account.deny"
	TypeAccountUpdate  Type = "account.update"
	TypeGroupCreate    Type = "group.create"
	TypeMsgPost        Type = "msg.post"
	TypeLoRaSent       Type = "lora.sent"
	TypeAlert          Type = "alert"
)

const (
	formatVersion = 1
	MaxBody       = 64 << 10 // bytes
	MaxPrio       = 9
	maxType       = 64
)

var (
	ErrBadSignature = errors.New("ledger: bad signature")
	ErrMalformed    = errors.New("ledger: malformed entry")
)

// Entry is one immutable record in the log.
type Entry struct {
	Key  Key
	TS   hlc.Timestamp
	Type Type
	Prio uint8 // 0-9, see decision 0015
	Body []byte
	Sig  []byte // ed25519 by Key.Origin over everything above
}

func (e *Entry) appendSigned(b []byte) []byte {
	b = append(b, formatVersion)
	b = append(b, e.Key.Origin[:]...)
	b = binary.AppendUvarint(b, e.Key.Seq)
	b = binary.AppendVarint(b, e.TS.Wall)
	b = binary.AppendUvarint(b, uint64(e.TS.Logical))
	b = appendBytes(b, []byte(e.Type))
	b = append(b, e.Prio)
	return appendBytes(b, e.Body)
}

// Sign sets Sig using the origin node's private key.
func (e *Entry) Sign(priv ed25519.PrivateKey) {
	e.Sig = ed25519.Sign(priv, e.appendSigned(nil))
}

// Verify checks that Sig was made by Key.Origin over the entry's contents.
func (e *Entry) Verify() error {
	if len(e.Sig) != ed25519.SignatureSize ||
		!ed25519.Verify(ed25519.PublicKey(e.Key.Origin[:]), e.appendSigned(nil), e.Sig) {
		return ErrBadSignature
	}
	return nil
}

// MarshalBinary encodes the entry for the wire and the database.
func (e *Entry) MarshalBinary() ([]byte, error) {
	return appendBytes(e.appendSigned(nil), e.Sig), nil
}

// UnmarshalBinary decodes an entry and checks its fields' limits (not its signature).
func (e *Entry) UnmarshalBinary(b []byte) error {
	r := reader{b: b}
	if r.byte() != formatVersion {
		return fmt.Errorf("%w: unknown version", ErrMalformed)
	}
	var out Entry
	copy(out.Key.Origin[:], r.fixed(len(out.Key.Origin)))
	out.Key.Seq = r.uvarint()
	out.TS.Wall = r.varint()
	logical := r.uvarint()
	out.Type = Type(r.bytes(maxType))
	out.Prio = r.byte()
	out.Body = r.bytes(MaxBody)
	out.Sig = r.bytes(ed25519.SignatureSize)
	switch {
	case r.err != nil:
		return fmt.Errorf("%w: %v", ErrMalformed, r.err)
	case len(r.b) != 0:
		return fmt.Errorf("%w: %d trailing bytes", ErrMalformed, len(r.b))
	case out.Key.Seq == 0:
		return fmt.Errorf("%w: seq 0", ErrMalformed)
	case out.Prio > MaxPrio:
		return fmt.Errorf("%w: priority %d", ErrMalformed, out.Prio)
	case logical > 1<<32-1:
		return fmt.Errorf("%w: logical clock overflow", ErrMalformed)
	}
	out.TS.Logical = uint32(logical)
	*e = out
	return nil
}

func appendBytes(b, v []byte) []byte {
	b = binary.AppendUvarint(b, uint64(len(v)))
	return append(b, v...)
}

// reader walks a byte slice; the first error sticks and later reads return zero values.
type reader struct {
	b   []byte
	err error
}

func (r *reader) fail(msg string) {
	if r.err == nil {
		r.err = errors.New(msg)
	}
}

func (r *reader) byte() byte {
	if r.err != nil || len(r.b) < 1 {
		r.fail("short read")
		return 0
	}
	v := r.b[0]
	r.b = r.b[1:]
	return v
}

func (r *reader) fixed(n int) []byte {
	if r.err != nil || len(r.b) < n {
		r.fail("short read")
		return make([]byte, n)
	}
	v := r.b[:n]
	r.b = r.b[n:]
	return v
}

func (r *reader) uvarint() uint64 {
	if r.err != nil {
		return 0
	}
	v, n := binary.Uvarint(r.b)
	if n <= 0 {
		r.fail("bad uvarint")
		return 0
	}
	r.b = r.b[n:]
	return v
}

func (r *reader) varint() int64 {
	if r.err != nil {
		return 0
	}
	v, n := binary.Varint(r.b)
	if n <= 0 {
		r.fail("bad varint")
		return 0
	}
	r.b = r.b[n:]
	return v
}

func (r *reader) bytes(max int) []byte {
	n := r.uvarint()
	if r.err != nil {
		return nil
	}
	if n > uint64(max) {
		r.fail("field too long")
		return nil
	}
	return append([]byte(nil), r.fixed(int(n))...)
}
```

```go
package ledger

import (
	"bytes"
	"crypto/sha256"
	"encoding/binary"
	"errors"
	"sort"
)

// Vector holds, per origin, the highest seq a node has with no gaps below it.
type Vector map[NodeID]uint64

// Range is a run of one origin's entries, From and To inclusive.
type Range struct {
	Origin   NodeID
	From, To uint64
}

// MaxRanges bounds a range list on the wire.
const MaxRanges = 1024

var errVector = errors.New("ledger: malformed vector or range list")

func (v Vector) sorted() []NodeID {
	ids := make([]NodeID, 0, len(v))
	for o, s := range v {
		if s > 0 {
			ids = append(ids, o)
		}
	}
	sort.Slice(ids, func(i, j int) bool { return bytes.Compare(ids[i][:], ids[j][:]) < 0 })
	return ids
}

// Hash summarises the vector; two nodes with equal hashes hold the same contiguous log.
func (v Vector) Hash() [32]byte {
	h := sha256.New()
	var buf [binary.MaxVarintLen64]byte
	for _, o := range v.sorted() {
		h.Write(o[:])
		h.Write(buf[:binary.PutUvarint(buf[:], v[o])])
	}
	var out [32]byte
	copy(out[:], h.Sum(nil))
	return out
}

// Missing lists what other has that v lacks, sorted by origin.
func (v Vector) Missing(other Vector) []Range {
	var out []Range
	for _, o := range other.sorted() {
		if other[o] > v[o] {
			out = append(out, Range{Origin: o, From: v[o] + 1, To: other[o]})
		}
	}
	return out
}

// AppendVector encodes v.
func AppendVector(b []byte, v Vector) []byte {
	ids := v.sorted()
	b = binary.AppendUvarint(b, uint64(len(ids)))
	for _, o := range ids {
		b = append(b, o[:]...)
		b = binary.AppendUvarint(b, v[o])
	}
	return b
}

// ParseVector decodes a vector and returns the bytes after it.
func ParseVector(b []byte) (Vector, []byte, error) {
	r := reader{b: b}
	n := r.uvarint()
	if r.err != nil || n > MaxRanges*64 {
		return nil, nil, errVector
	}
	v := make(Vector, n)
	for i := uint64(0); i < n; i++ {
		var o NodeID
		copy(o[:], r.fixed(len(o)))
		v[o] = r.uvarint()
	}
	if r.err != nil {
		return nil, nil, errVector
	}
	return v, r.b, nil
}

// AppendRanges encodes a range list.
func AppendRanges(b []byte, rs []Range) []byte {
	b = binary.AppendUvarint(b, uint64(len(rs)))
	for _, x := range rs {
		b = append(b, x.Origin[:]...)
		b = binary.AppendUvarint(b, x.From)
		b = binary.AppendUvarint(b, x.To)
	}
	return b
}

// ParseRanges decodes a range list and returns the bytes after it.
func ParseRanges(b []byte) ([]Range, []byte, error) {
	r := reader{b: b}
	n := r.uvarint()
	if r.err != nil || n > MaxRanges {
		return nil, nil, errVector
	}
	rs := make([]Range, 0, n)
	for i := uint64(0); i < n; i++ {
		var x Range
		copy(x.Origin[:], r.fixed(len(x.Origin)))
		x.From, x.To = r.uvarint(), r.uvarint()
		rs = append(rs, x)
	}
	if r.err != nil {
		return nil, nil, errVector
	}
	return rs, r.b, nil
}
```

```go
package ledger

import "errors"

// ErrNotFound is returned by Get for an entry the store doesn't hold.
var ErrNotFound = errors.New("ledger: entry not found")

// Store keeps entries. Callers verify signatures before Put.
type Store interface {
	// Put stores e; added is false if the store already had it.
	Put(e *Entry) (added bool, err error)
	Get(k Key) (*Entry, error)
	// Vector is the highest contiguous seq per origin.
	Vector() (Vector, error)
	// Range returns the stored entries in r, ascending by seq; gaps are skipped.
	Range(r Range) ([]*Entry, error)
	// All returns every entry ordered by timestamp, then origin, then seq.
	All() ([]*Entry, error)
	Close() error
}
```

```go
package ledger

import (
	"bytes"
	"sort"
	"sync"
)

// MemStore is a Store in memory, for tests and the mock village.
type MemStore struct {
	mu      sync.Mutex
	entries map[Key]*Entry
	contig  Vector
}

// NewMemStore returns an empty MemStore.
func NewMemStore() *MemStore {
	return &MemStore{entries: map[Key]*Entry{}, contig: Vector{}}
}

func (s *MemStore) Put(e *Entry) (bool, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	if _, ok := s.entries[e.Key]; ok {
		return false, nil
	}
	s.entries[e.Key] = e
	for {
		next := Key{Origin: e.Key.Origin, Seq: s.contig[e.Key.Origin] + 1}
		if _, ok := s.entries[next]; !ok {
			break
		}
		s.contig[e.Key.Origin] = next.Seq
	}
	return true, nil
}

func (s *MemStore) Get(k Key) (*Entry, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	if e, ok := s.entries[k]; ok {
		return e, nil
	}
	return nil, ErrNotFound
}

func (s *MemStore) Vector() (Vector, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	v := make(Vector, len(s.contig))
	for o, n := range s.contig {
		v[o] = n
	}
	return v, nil
}

func (s *MemStore) Range(r Range) ([]*Entry, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	var out []*Entry
	for seq := r.From; seq <= r.To && seq >= r.From; seq++ {
		if e, ok := s.entries[Key{Origin: r.Origin, Seq: seq}]; ok {
			out = append(out, e)
		}
	}
	return out, nil
}

func (s *MemStore) All() ([]*Entry, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	out := make([]*Entry, 0, len(s.entries))
	for _, e := range s.entries {
		out = append(out, e)
	}
	SortEntries(out)
	return out, nil
}

func (s *MemStore) Close() error { return nil }

// SortEntries orders entries by timestamp, then origin, then seq.
func SortEntries(es []*Entry) {
	sort.Slice(es, func(i, j int) bool {
		a, b := es[i], es[j]
		if a.TS != b.TS {
			return a.TS.Less(b.TS)
		}
		if c := bytes.Compare(a.Key.Origin[:], b.Key.Origin[:]); c != 0 {
			return c < 0
		}
		return a.Key.Seq < b.Key.Seq
	})
}
```

```go
// Package sqlstore is a ledger.Store in SQLite (pure Go, no cgo).
package sqlstore

import (
	"database/sql"
	"errors"
	"fmt"

	"github.com/johnusher/batnet/internal/ledger"
	_ "modernc.org/sqlite"
)

const schema = `
CREATE TABLE IF NOT EXISTS entries (
	origin  BLOB    NOT NULL,
	seq     INTEGER NOT NULL,
	wall    INTEGER NOT NULL,
	logical INTEGER NOT NULL,
	raw     BLOB    NOT NULL,
	PRIMARY KEY (origin, seq)
);
CREATE INDEX IF NOT EXISTS entries_clock ON entries (wall, logical, origin, seq);
CREATE TABLE IF NOT EXISTS contig (
	origin BLOB PRIMARY KEY,
	seq    INTEGER NOT NULL
);`

// Store is a ledger.Store backed by one SQLite file.
type Store struct{ db *sql.DB }

// Open opens or creates the database at path.
func Open(path string) (*Store, error) {
	db, err := sql.Open("sqlite", "file:"+path+"?_pragma=journal_mode(WAL)&_pragma=busy_timeout(5000)&_pragma=synchronous(NORMAL)")
	if err != nil {
		return nil, err
	}
	db.SetMaxOpenConns(1) // one writer; SQLite serialises anyway
	if _, err := db.Exec(schema); err != nil {
		db.Close()
		return nil, fmt.Errorf("sqlstore: schema: %w", err)
	}
	return &Store{db: db}, nil
}

func (s *Store) Close() error { return s.db.Close() }

func (s *Store) Put(e *ledger.Entry) (bool, error) {
	raw, err := e.MarshalBinary()
	if err != nil {
		return false, err
	}
	tx, err := s.db.Begin()
	if err != nil {
		return false, err
	}
	defer tx.Rollback()
	res, err := tx.Exec(`INSERT OR IGNORE INTO entries (origin, seq, wall, logical, raw) VALUES (?, ?, ?, ?, ?)`,
		e.Key.Origin[:], int64(e.Key.Seq), e.TS.Wall, int64(e.TS.Logical), raw)
	if err != nil {
		return false, err
	}
	if n, _ := res.RowsAffected(); n == 0 {
		return false, nil
	}
	var c int64
	err = tx.QueryRow(`SELECT seq FROM contig WHERE origin = ?`, e.Key.Origin[:]).Scan(&c)
	if err != nil && !errors.Is(err, sql.ErrNoRows) {
		return false, err
	}
	for {
		var one int
		err := tx.QueryRow(`SELECT 1 FROM entries WHERE origin = ? AND seq = ?`, e.Key.Origin[:], c+1).Scan(&one)
		if errors.Is(err, sql.ErrNoRows) {
			break
		}
		if err != nil {
			return false, err
		}
		c++
	}
	if _, err := tx.Exec(`INSERT INTO contig (origin, seq) VALUES (?, ?) ON CONFLICT (origin) DO UPDATE SET seq = excluded.seq`,
		e.Key.Origin[:], c); err != nil {
		return false, err
	}
	return true, tx.Commit()
}

func (s *Store) Get(k ledger.Key) (*ledger.Entry, error) {
	var raw []byte
	err := s.db.QueryRow(`SELECT raw FROM entries WHERE origin = ? AND seq = ?`, k.Origin[:], int64(k.Seq)).Scan(&raw)
	if errors.Is(err, sql.ErrNoRows) {
		return nil, ledger.ErrNotFound
	}
	if err != nil {
		return nil, err
	}
	return decode(raw)
}

func (s *Store) Vector() (ledger.Vector, error) {
	rows, err := s.db.Query(`SELECT origin, seq FROM contig WHERE seq > 0`)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	v := ledger.Vector{}
	for rows.Next() {
		var o []byte
		var seq int64
		if err := rows.Scan(&o, &seq); err != nil {
			return nil, err
		}
		var id ledger.NodeID
		copy(id[:], o)
		v[id] = uint64(seq)
	}
	return v, rows.Err()
}

func (s *Store) Range(r ledger.Range) ([]*ledger.Entry, error) {
	return s.query(`SELECT raw FROM entries WHERE origin = ? AND seq BETWEEN ? AND ? ORDER BY seq`,
		r.Origin[:], int64(r.From), int64(r.To))
}

func (s *Store) All() ([]*ledger.Entry, error) {
	out, err := s.query(`SELECT raw FROM entries ORDER BY wall, logical, origin, seq`)
	if err == nil {
		ledger.SortEntries(out) // same tie-break as MemStore
	}
	return out, err
}

func (s *Store) query(q string, args ...any) ([]*ledger.Entry, error) {
	rows, err := s.db.Query(q, args...)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []*ledger.Entry
	for rows.Next() {
		var raw []byte
		if err := rows.Scan(&raw); err != nil {
			return nil, err
		}
		e, err := decode(raw)
		if err != nil {
			return nil, err
		}
		out = append(out, e)
	}
	return out, rows.Err()
}

func decode(raw []byte) (*ledger.Entry, error) {
	var e ledger.Entry
	if err := e.UnmarshalBinary(raw); err != nil {
		return nil, err
	}
	return &e, nil
}
```

```go
import (
	"crypto/ed25519"
	"encoding/json"
	"fmt"
	"sync"

	"github.com/johnusher/batnet/internal/hlc"

// Author appends this node's own entries to the log.
type Author struct {
	mu    sync.Mutex
	priv  ed25519.PrivateKey
	id    NodeID
	clock *hlc.Clock
	store Store
	seq   uint64
}

// NewAuthor continues from the highest seq of this node already in store.
func NewAuthor(priv ed25519.PrivateKey, clock *hlc.Clock, store Store) (*Author, error) {
	id := NodeIDFromPublic(priv.Public().(ed25519.PublicKey))
	v, err := store.Vector()
	if err != nil {
		return nil, err
	}
	return &Author{priv: priv, id: id, clock: clock, store: store, seq: v[id]}, nil
}

// ID is this node's identity.
func (a *Author) ID() NodeID { return a.id }

// Append signs and stores a new entry with body encoded as JSON.
func (a *Author) Append(t Type, prio uint8, body any) (*Entry, error) {
	if prio > MaxPrio {
		return nil, fmt.Errorf("ledger: priority %d above %d", prio, MaxPrio)
	}
	raw, err := json.Marshal(body)
	if err != nil {
		return nil, err
	}
	if len(raw) > MaxBody {
		return nil, fmt.Errorf("ledger: body of %d bytes above %d", len(raw), MaxBody)
	}
	a.mu.Lock()
	defer a.mu.Unlock()
	e := &Entry{Key: Key{Origin: a.id, Seq: a.seq + 1}, TS: a.clock.Now(), Type: t, Prio: prio, Body: raw}
	e.Sign(a.priv)
	if _, err := a.store.Put(e); err != nil {
		return nil, err
	}
	a.seq++
	return e, nil
}
```

```go
// Package seal encrypts traffic between a village's nodes with keys derived
// from the village key (XChaCha20-Poly1305, HKDF-SHA256).
package seal

import (
	"crypto/cipher"
	"crypto/rand"
	"crypto/sha256"
	"errors"
	"io"

	"golang.org/x/crypto/chacha20poly1305"
	"golang.org/x/crypto/hkdf"
)

// PurposeMesh is the key purpose for traffic over the WiFi mesh.
const PurposeMesh = "batnet mesh v1"

// ErrOpen means the box wasn't made with this village's key, or was damaged.
var ErrOpen = errors.New("seal: cannot open (wrong village key or damaged)")

// VillageKey is the shared secret every house in a village holds.
type VillageKey [32]byte

// NewVillageKey makes a random key for a new village.
func NewVillageKey() (VillageKey, error) {
	var k VillageKey
	_, err := rand.Read(k[:])
	return k, err
}

// Derive returns a 32-byte key for one purpose, so keys for the mesh, LoRa and
// the like never coincide.
func (k VillageKey) Derive(purpose string) [32]byte {
	var out [32]byte
	if _, err := io.ReadFull(hkdf.New(sha256.New, k[:], nil, []byte(purpose)), out[:]); err != nil {
		panic(err) // HKDF-SHA256 can always produce 32 bytes
	}
	return out
}

// Sealer seals and opens boxes for one purpose.
type Sealer struct{ aead cipher.AEAD }

// NewSealer returns a Sealer for k and purpose.
func NewSealer(k VillageKey, purpose string) (*Sealer, error) {
	key := k.Derive(purpose)
	aead, err := chacha20poly1305.NewX(key[:])
	if err != nil {
		return nil, err
	}
	return &Sealer{aead: aead}, nil
}

// Seal returns nonce || ciphertext.
func (s *Sealer) Seal(plain []byte) ([]byte, error) {
	nonce := make([]byte, chacha20poly1305.NonceSizeX, chacha20poly1305.NonceSizeX+len(plain)+s.aead.Overhead())
	if _, err := rand.Read(nonce); err != nil {
		return nil, err
	}
	return s.aead.Seal(nonce, nonce, plain, nil), nil
}

// Open reverses Seal.
func (s *Sealer) Open(box []byte) ([]byte, error) {
	if len(box) < chacha20poly1305.NonceSizeX+s.aead.Overhead() {
		return nil, ErrOpen
	}
	plain, err := s.aead.Open(nil, box[:chacha20poly1305.NonceSizeX], box[chacha20poly1305.NonceSizeX:], nil)
	if err != nil {
		return nil, ErrOpen
	}
	return plain, nil
}
```

```go
package meshsync

import (
	"encoding/binary"
	"errors"

	"github.com/johnusher/batnet/internal/ledger"
)

// Packet kinds: the first byte of every (unsealed) packet.
const (
	kindEntry     byte = 'E' // broadcast: one new entry
	kindHeartbeat byte = 'H' // broadcast: sender address and vector hash
	kindVectorReq byte = 'V' // request: send me your vector
	kindRangeReq  byte = 'R' // request: send me these ranges
)

const (
	maxEntriesPerResponse = 500
	maxAddr               = 64
)

var errWire = errors.New("meshsync: malformed packet")

type heartbeat struct {
	Addr string
	Hash [32]byte
}

func encodeEntry(e *ledger.Entry) []byte {
	raw, _ := e.MarshalBinary()
	return append([]byte{kindEntry}, raw...)
}

func encodeHeartbeat(h heartbeat) []byte {
	b := []byte{kindHeartbeat}
	b = binary.AppendUvarint(b, uint64(len(h.Addr)))
	b = append(b, h.Addr...)
	return append(b, h.Hash[:]...)
}

// decodeBroadcast returns exactly one of an entry or a heartbeat.
func decodeBroadcast(b []byte) (*ledger.Entry, *heartbeat, error) {
	if len(b) < 2 {
		return nil, nil, errWire
	}
	switch b[0] {
	case kindEntry:
		var e ledger.Entry
		if err := e.UnmarshalBinary(b[1:]); err != nil {
			return nil, nil, errWire
		}
		return &e, nil, nil
	case kindHeartbeat:
		n, k := binary.Uvarint(b[1:])
		rest := b[1:]
		if k <= 0 || n > maxAddr || uint64(len(rest)-k) != n+32 {
			return nil, nil, errWire
		}
		rest = rest[k:]
		h := heartbeat{Addr: string(rest[:n])}
		copy(h.Hash[:], rest[n:])
		return nil, &h, nil
	}
	return nil, nil, errWire
}

func encodeEntries(es []*ledger.Entry) []byte {
	b := binary.AppendUvarint(nil, uint64(len(es)))
	for _, e := range es {
		raw, _ := e.MarshalBinary()
		b = binary.AppendUvarint(b, uint64(len(raw)))
		b = append(b, raw...)
	}
	return b
}

func decodeEntries(b []byte) ([]*ledger.Entry, error) {
	n, k := binary.Uvarint(b)
	if k <= 0 || n > maxEntriesPerResponse {
		return nil, errWire
	}
	b = b[k:]
	out := make([]*ledger.Entry, 0, n)
	for i := uint64(0); i < n; i++ {
		l, k := binary.Uvarint(b)
		if k <= 0 || uint64(len(b)-k) < l {
			return nil, errWire
		}
		var e ledger.Entry
		if err := e.UnmarshalBinary(b[k : k+int(l)]); err != nil {
			return nil, errWire
		}
		out = append(out, &e)
		b = b[k+int(l):]
	}
	if len(b) != 0 {
		return nil, errWire
	}
	return out, nil
}
```

```go
// Package mock is an in-process stand-in for the batman-adv mesh: nodes,
// links between them, power cuts and packet loss, so the sync code runs
// without radios.
package mock

import (
	"context"
	"errors"
	"math/rand"
	"sync"

	"github.com/johnusher/batnet/internal/meshsync"
)

// ErrUnreachable means no mesh path to that node right now.
var ErrUnreachable = errors.New("mock: node unreachable")

const inboxSize = 4096

// Mesh connects Links like batman-adv: a broadcast floods every node that has
// a path to the sender.
type Mesh struct {
	mu    sync.Mutex
	links map[string]*Link
	adj   map[string]map[string]bool
	down  map[string]bool
	loss  float64
	rnd   *rand.Rand
}

// NewMesh returns an empty mesh; seed fixes the loss pattern.
func NewMesh(seed int64) *Mesh {
	return &Mesh{links: map[string]*Link{}, adj: map[string]map[string]bool{}, down: map[string]bool{}, rnd: rand.New(rand.NewSource(seed))}
}

// Add creates a node.
func (m *Mesh) Add(addr string) *Link {
	m.mu.Lock()
	defer m.mu.Unlock()
	l := &Link{mesh: m, addr: addr, inbox: make(chan meshsync.Packet, inboxSize)}
	m.links[addr] = l
	m.adj[addr] = map[string]bool{}
	return l
}

// Connect puts two nodes in radio range of each other.
func (m *Mesh) Connect(a, b string) { m.setEdge(a, b, true) }

// Disconnect takes two nodes out of range.
func (m *Mesh) Disconnect(a, b string) { m.setEdge(a, b, false) }

func (m *Mesh) setEdge(a, b string, on bool) {
	m.mu.Lock()
	defer m.mu.Unlock()
	if on {
		m.adj[a][b], m.adj[b][a] = true, true
	} else {
		delete(m.adj[a], b)
		delete(m.adj[b], a)
	}
}

// SetUp cuts or restores a node's power.
func (m *Mesh) SetUp(addr string, up bool) {
	m.mu.Lock()
	defer m.mu.Unlock()
	m.down[addr] = !up
}

// SetLoss sets the chance that a broadcast misses each receiver.
func (m *Mesh) SetLoss(p float64) {
	m.mu.Lock()
	defer m.mu.Unlock()
	m.loss = p
}

// reachable lists the powered nodes with a path from addr, excluding addr. Caller holds mu.
func (m *Mesh) reachable(addr string) []string {
	if m.down[addr] {
		return nil
	}
	seen := map[string]bool{addr: true}
	queue := []string{addr}
	var out []string
	for len(queue) > 0 {
		x := queue[0]
		queue = queue[1:]
		for y := range m.adj[x] {
			if seen[y] || m.down[y] {
				continue
			}
			seen[y] = true
			out = append(out, y)
			queue = append(queue, y)
		}
	}
	return out
}

// Link is one node's connection to the mock mesh.
type Link struct {
	mesh    *Mesh
	addr    string
	inbox   chan meshsync.Packet
	hmu     sync.Mutex
	handler func(from string, req []byte) []byte
}

func (l *Link) Addr() string                    { return l.addr }
func (l *Link) Inbox() <-chan meshsync.Packet { return l.inbox }

func (l *Link) SetHandler(h func(from string, req []byte) []byte) {
	l.hmu.Lock()
	defer l.hmu.Unlock()
	l.handler = h
}

func (l *Link) Broadcast(pkt []byte) error {
	m := l.mesh
	m.mu.Lock()
	var targets []*Link
	for _, addr := range m.reachable(l.addr) {
		if m.loss > 0 && m.rnd.Float64() < m.loss {
			continue
		}
		targets = append(targets, m.links[addr])
	}
	m.mu.Unlock()
	for _, t := range targets {
		select {
		case t.inbox <- meshsync.Packet{From: l.addr, Data: append([]byte(nil), pkt...)}:
		default: // a full inbox drops the packet, like a busy radio
		}
	}
	return nil
}

func (l *Link) Request(ctx context.Context, to string, req []byte) ([]byte, error) {
	m := l.mesh
	m.mu.Lock()
	ok := false
	for _, addr := range m.reachable(l.addr) {
		if addr == to {
			ok = true
			break
		}
	}
	target := m.links[to]
	m.mu.Unlock()
	if !ok || target == nil {
		return nil, ErrUnreachable
	}
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	target.hmu.Lock()
	h := target.handler
	target.hmu.Unlock()
	if h == nil {
		return nil, ErrUnreachable
	}
	return h(l.addr, append([]byte(nil), req...)), nil
}
```

```go
func newVillage(t *testing.T, m *mock.Mesh, key seal.VillageKey, n int) []*house {

	t.Cleanup(cancel)

		h := &house{addr: fmt.Sprintf("house-%d", i), store: ledger.NewMemStore()}
		h.link = m.Add(h.addr)
		_, priv, _ := ed25519.GenerateKey(nil)
		clock := hlc.New(func() int64 { return time.Now().UnixMilli() })
		h.author, _ = ledger.NewAuthor(priv, clock, h.store)
		sealer, _ := seal.NewSealer(key, seal.PurposeMesh)
		h.sync = meshsync.New(meshsync.Config{
			Link: h.link, Store: h.store, Clock: clock, Sealer: sealer,
			Heartbeat: 20 * time.Millisecond,
			OnEntry:   func(*ledger.Entry) { h.got.Add(1) },

// waitFor polls until every house in hs holds want entries.
func waitFor(t *testing.T, hs []*house, want int) {

	deadline := time.Now().Add(5 * time.Second)

			if h.count() != want {

		}
		time.Sleep(10 * time.Millisecond)
	}
	for _, h := range hs {

	t.FailNow()
}
```

```go
package meshsync

import (
	"context"
	"sync"
	"time"

	"github.com/johnusher/batnet/internal/hlc"
	"github.com/johnusher/batnet/internal/ledger"
	"github.com/johnusher/batnet/internal/seal"
)

// Config wires a Syncer to its node.
type Config struct {
	Link      Link
	Store     ledger.Store
	Clock     *hlc.Clock
	Sealer    *seal.Sealer
	Heartbeat time.Duration       // 5 s on a Pi
	OnEntry   func(*ledger.Entry) // once per entry new to this node, from the network
	Logf      func(format string, args ...any)
}

// Syncer keeps one node's log in step with the mesh.
type Syncer struct {
	cfg       Config
	mu        sync.Mutex
	repairing map[string]bool
}

const (
	maxRangesPerRequest = 64
	maxRepairRounds     = 50
	requestTimeout      = 10 * time.Second
)

// New returns a Syncer and installs its request handler on the link.
func New(cfg Config) *Syncer {
	if cfg.Heartbeat == 0 {
		cfg.Heartbeat = 5 * time.Second
	}
	if cfg.OnEntry == nil {
		cfg.OnEntry = func(*ledger.Entry) {}
	}
	if cfg.Logf == nil {
		cfg.Logf = func(string, ...any) {}
	}
	s := &Syncer{cfg: cfg, repairing: map[string]bool{}}
	cfg.Link.SetHandler(s.handle)
	return s
}

// Publish broadcasts an entry this node just appended.
func (s *Syncer) Publish(e *ledger.Entry) error { return s.broadcast(encodeEntry(e)) }

// SendHeartbeat broadcasts this node's address and vector hash.
func (s *Syncer) SendHeartbeat() error {
	v, err := s.cfg.Store.Vector()
	if err != nil {
		return err
	}
	return s.broadcast(encodeHeartbeat(heartbeat{Addr: s.cfg.Link.Addr(), Hash: v.Hash()}))
}

func (s *Syncer) broadcast(pkt []byte) error {
	box, err := s.cfg.Sealer.Seal(pkt)
	if err != nil {
		return err
	}
	return s.cfg.Link.Broadcast(box)
}

// Run handles broadcasts and sends heartbeats until ctx ends.
func (s *Syncer) Run(ctx context.Context) {
	tick := time.NewTicker(s.cfg.Heartbeat)
	defer tick.Stop()
	s.SendHeartbeat()
	for {
		select {
		case <-ctx.Done():
			return
		case <-tick.C:
			if err := s.SendHeartbeat(); err != nil {
				s.cfg.Logf("heartbeat: %v", err)
			}
		case p := <-s.cfg.Link.Inbox():
			s.receive(ctx, p)
		}
	}
}

func (s *Syncer) receive(ctx context.Context, p Packet) {
	plain, err := s.cfg.Sealer.Open(p.Data)
	if err != nil {
		return // another village, or noise
	}
	e, hb, err := decodeBroadcast(plain)
	if err != nil {
		return
	}
	if e != nil {
		s.accept(e)
		return
	}
	if hb.Addr == s.cfg.Link.Addr() {
		return
	}
	v, err := s.cfg.Store.Vector()
	if err != nil || v.Hash() == hb.Hash {
		return
	}
	s.mu.Lock()
	busy := s.repairing[hb.Addr]
	s.repairing[hb.Addr] = true
	s.mu.Unlock()
	if busy {
		return
	}
	go func() {
		defer func() { s.mu.Lock(); delete(s.repairing, hb.Addr); s.mu.Unlock() }()
		if n, err := s.Repair(ctx, hb.Addr); err != nil {
			s.cfg.Logf("repair from %s: %v", hb.Addr, err)
		} else if n > 0 {
			s.cfg.Logf("repair from %s: %d entries", hb.Addr, n)
		}
	}()
}

// accept verifies and stores an entry from the network; true if it was new.
func (s *Syncer) accept(e *ledger.Entry) bool {
	if e.Verify() != nil {
		return false
	}
	added, err := s.cfg.Store.Put(e)
	if err != nil || !added {
		return false
	}
	s.cfg.Clock.Observe(e.TS)
	s.cfg.OnEntry(e)
	return true
}

// Repair pulls from peer whatever it has that this node lacks.
func (s *Syncer) Repair(ctx context.Context, peer string) (int, error) {
	total := 0
	for round := 0; round < maxRepairRounds; round++ {
		theirs, err := ask(s, ctx, peer, []byte{kindVectorReq}, func(b []byte) (ledger.Vector, error) {
			v, rest, err := ledger.ParseVector(b)
			if err == nil && len(rest) != 0 {
				err = errWire
			}
			return v, err
		})
		if err != nil {
			return total, err
		}
		mine, err := s.cfg.Store.Vector()
		if err != nil {
			return total, err
		}
		missing := mine.Missing(theirs)
		if len(missing) == 0 {
			return total, nil
		}
		if len(missing) > maxRangesPerRequest {
			missing = missing[:maxRangesPerRequest]
		}
		es, err := ask(s, ctx, peer, ledger.AppendRanges([]byte{kindRangeReq}, missing), decodeEntries)
		if err != nil {
			return total, err
		}
		added := 0
		for _, e := range es {
			if s.accept(e) {
				added++
			}
		}
		total += added
		if added == 0 {
			return total, nil // nothing usable came back; try again at the next heartbeat
		}
	}
	return total, nil
}

// ask sends a sealed request and decodes the sealed answer. (A function, not a
// method: Go methods can't have type parameters.)
func ask[T any](s *Syncer, ctx context.Context, peer string, req []byte, decode func([]byte) (T, error)) (T, error) {
	var zero T
	box, err := s.cfg.Sealer.Seal(req)
	if err != nil {
		return zero, err
	}
	ctx, cancel := context.WithTimeout(ctx, requestTimeout)
	defer cancel()
	resp, err := s.cfg.Link.Request(ctx, peer, box)
	if err != nil {
		return zero, err
	}
	plain, err := s.cfg.Sealer.Open(resp)
	if err != nil {
		return zero, err
	}
	return decode(plain)
}

// handle answers another node's request; nil means no answer.
func (s *Syncer) handle(from string, req []byte) []byte {
	plain, err := s.cfg.Sealer.Open(req)
	if err != nil || len(plain) == 0 {
		return nil
	}
	var resp []byte
	switch plain[0] {
	case kindVectorReq:
		v, err := s.cfg.Store.Vector()
		if err != nil {
			return nil
		}
		resp = ledger.AppendVector(nil, v)
	case kindRangeReq:
		rs, rest, err := ledger.ParseRanges(plain[1:])
		if err != nil || len(rest) != 0 {
			return nil
		}
		var out []*ledger.Entry
		for _, r := range rs {
			es, err := s.cfg.Store.Range(r)
			if err != nil {
				return nil
			}
			for _, e := range es {
				if len(out) == maxEntriesPerResponse {
					break
				}
				out = append(out, e)
			}
		}
		resp = encodeEntries(out)
	default:
		return nil
	}
	box, err := s.cfg.Sealer.Seal(resp)
	if err != nil {
		return nil
	}
	return box
}
```

```go
func runMock(n int, loss float64, dur time.Duration) error {
	key, err := seal.NewVillageKey()
	if err != nil {
		return err
	}
	m := mock.NewMesh(time.Now().UnixNano())
	m.SetLoss(loss)
	ctx, cancel := context.WithTimeout(context.Background(), dur)
	defer cancel()
	start := time.Now()
	type node struct {
		name   string
		author *ledger.Author
		sync   *meshsync.Syncer
		store  *ledger.MemStore
	}
	var nodes []*node
	for i := 0; i < n; i++ {
		name := houseNames[i%len(houseNames)]
		if i >= len(houseNames) {
			name = fmt.Sprintf("%s-%d", name, i/len(houseNames)+1)
		}
		nd := &node{name: name, store: ledger.NewMemStore()}
		_, priv, err := ed25519.GenerateKey(nil)
		if err != nil {
			return err
		}
		clock := hlc.New(func() int64 { return time.Now().UnixMilli() })
		if nd.author, err = ledger.NewAuthor(priv, clock, nd.store); err != nil {
			return err
		}
		sealer, err := seal.NewSealer(key, seal.PurposeMesh)
		if err != nil {
			return err
		}
		link := m.Add(name)
		nd.sync = meshsync.New(meshsync.Config{
			Link: link, Store: nd.store, Clock: clock, Sealer: sealer,
			Heartbeat: 500 * time.Millisecond,
			OnEntry: func(e *ledger.Entry) {
				lag := time.Since(time.UnixMilli(e.TS.Wall)).Round(time.Millisecond)
				fmt.Printf("%6s  %-10s got  %s  (after %v)\n", time.Since(start).Round(100*time.Millisecond), name, e.Body, lag)
			},
		})
		if i > 0 {
			m.Connect(nodes[i-1].name, name)
		}
		go nd.sync.Run(ctx)
		nodes = append(nodes, nd)
	}

	defer tick.Stop()
	for i := 0; ; i++ {

		case <-ctx.Done():
			fmt.Println("\nWhat each house holds:")

			nd.sync.Publish(e)
		}
	}
}
```
